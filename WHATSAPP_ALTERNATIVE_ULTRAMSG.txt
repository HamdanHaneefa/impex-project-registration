function sendWhatsAppMessage(data) {
  // UltraMsg Configuration
  const ULTRAMSG_INSTANCE = 'YOUR_INSTANCE_ID'; // Replace with your instance ID
  const ULTRAMSG_TOKEN = 'YOUR_TOKEN'; // Replace with your token
  
  // Clean phone number
  const cleanPhone = data.phone.replace(/\s/g, '').replace('+', '');
  
  // Create message
  const message = `🎉 *Registration Confirmed!*

Dear ${data.fullName},

Your registration for *IFPD Meet 2026* is confirmed!

📅 *Event Details:*
Date: Saturday, 9 May 2026
Time: 10:00 AM – 3:00 PM

• Keynote by Mr. Renjith Kesav
• Network with 100+ leaders
• Complimentary meals

Need help? Call +91 97786 65499

See you at the event!
Team Impex`;

  // UltraMsg API endpoint
  const apiUrl = `https://api.ultramsg.com/${ULTRAMSG_INSTANCE}/messages/chat`;
  
  // Prepare payload
  const payload = {
    token: ULTRAMSG_TOKEN,
    to: cleanPhone,
    body: message
  };
  
  // Send request
  const options = {
    method: 'post',
    contentType: 'application/x-www-form-urlencoded',
    payload: payload,
    muteHttpExceptions: true
  };
  
  const response = UrlFetchApp.fetch(apiUrl, options);
  const responseCode = response.getResponseCode();
  
  if (responseCode !== 200) {
    throw new Error('WhatsApp API error: ' + response.getContentText());
  }
  
  return true;
}
